import hashlib
import importlib.util
import json
import os
from pathlib import Path
import shutil
import unittest
from unittest.mock import patch

BASE = Path(__file__).resolve().parent
MODULE = BASE.parent / 'save-memory.py'

def encoded(revision=0, entries=None, receipts=None, history=None):
    data = dict(format_version=1, role_key='content-creator', revision=revision,
                entries=entries or [], receipts=receipts or [], history=history or [])
    return ('# 岗位记忆\n\n```json\n' + json.dumps(data, ensure_ascii=False, indent=2) + '\n```\n').encode()

spec = importlib.util.spec_from_file_location('save_memory_checked', MODULE)
saver = importlib.util.module_from_spec(spec)
spec.loader.exec_module(saver)

class SaveChecks(unittest.TestCase):
    def setUp(self):
        self.root = BASE / self._testMethodName
        if self.root.exists():
            shutil.rmtree(self.root)
        self.root.mkdir()
        self.rel = 'memory.md'
        self.target = self.root / self.rel
        self.old = encoded(0)
        self.new = encoded(1, history=[dict(revision=0, entries=[], receipts=[])])
        self.target.write_bytes(self.old)
        self.allowed = {self.rel, self.rel + '.pending'}
        self.digest = hashlib.sha256(self.old).hexdigest()

    def call(self, **changes):
        args = dict(root=self.root, target_rel=self.rel, candidate_bytes=self.new,
                    expected_sha256=self.digest, allowed_paths=self.allowed)
        args.update(changes)
        return saver.save_memory(**args)

    def test_save_and_no_side_files(self):
        result = self.call()
        self.assertEqual(result['status'], 'saved', result)
        self.assertEqual(self.target.read_bytes(), self.new)
        self.assertFalse((self.root / (self.rel + '.pending')).exists())
        self.assertEqual({p.name for p in self.root.iterdir()}, {self.rel})

    def test_first_archive_creation_and_noop(self):
        rel = 'archive/new-role.md'
        result = self.call(target_rel=rel, expected_sha256=None,
                           allowed_paths={rel, rel + '.pending'})
        self.assertEqual(result['status'], 'saved', result)
        self.assertEqual((self.root / rel).read_bytes(), self.new)
        result = self.call(target_rel=rel, candidate_bytes=self.new,
                           expected_sha256=hashlib.sha256(self.new).hexdigest(),
                           allowed_paths={rel, rel + '.pending'})
        self.assertEqual(result['status'], 'unchanged', result)
        self.assertFalse((self.root / (rel + '.pending')).exists())

    def test_pending_permission_missing(self):
        self.assertEqual(self.call(allowed_paths={self.rel})['status'], 'pending')
        self.assertEqual(self.target.read_bytes(), self.old)
        self.assertFalse((self.root / (self.rel + '.pending')).exists())

    def test_invalid_candidate_does_not_create_archive_directory(self):
        rel = 'archive/bad.md'
        result = self.call(target_rel=rel, candidate_bytes=b'{broken', expected_sha256=None,
                           allowed_paths={rel, rel + '.pending'})
        self.assertEqual(result['status'], 'pending', result)
        self.assertFalse((self.root / 'archive').exists())

    def test_original_absence_is_explicit(self):
        self.assertEqual(self.call(expected_sha256=None)['status'], 'pending')
        self.target.unlink()
        self.assertEqual(self.call()['status'], 'pending')
        self.assertFalse(self.target.exists())

    def test_stale_hash(self):
        result = self.call(expected_sha256='0' * 64)
        self.assertEqual(result['status'], 'pending', result)
        self.assertEqual(self.target.read_bytes(), self.old)

    def test_existing_pending_preserved(self):
        pending = self.root / (self.rel + '.pending')
        pending.write_bytes(b'incomplete candidate')
        result = self.call()
        self.assertEqual(result['status'], 'pending', result)
        self.assertEqual(pending.read_bytes(), b'incomplete candidate')
        self.assertEqual(self.target.read_bytes(), self.old)

    def test_pending_symlink_preserved(self):
        pending = self.root / (self.rel + '.pending')
        pending.symlink_to(self.target)
        result = self.call()
        self.assertEqual(result['status'], 'pending', result)
        self.assertTrue(pending.is_symlink())
        self.assertEqual(self.target.read_bytes(), self.old)

    def test_escape_and_aliases(self):
        for rel in ('../escape.md', '/escape.md', './memory.md', 'x/../memory.md'):
            with self.subTest(rel=rel):
                result = self.call(target_rel=rel, allowed_paths={rel, rel + '.pending'})
                self.assertEqual(result['status'], 'pending', result)
        for allowed in (self.allowed | {'MEMORY.md'}, self.allowed | {'memory.md-bridge', 'memory.md/child.md'}):
            self.assertEqual(self.call(allowed_paths=allowed)['status'], 'pending')
        alias = self.root / 'alias.md'
        os.link(self.target, alias)
        self.assertEqual(self.call(allowed_paths=self.allowed | {'alias.md'})['status'], 'pending')
        self.assertEqual(self.target.read_bytes(), self.old)

    def test_symlink_parent(self):
        real = self.root / 'real'
        real.mkdir()
        (self.root / 'linked').symlink_to(real, target_is_directory=True)
        rel = 'linked/new.md'
        result = self.call(target_rel=rel, expected_sha256=None, allowed_paths={rel, rel + '.pending'})
        self.assertEqual(result['status'], 'pending', result)
        self.assertFalse((real / 'new.md').exists())

    def test_unknown_format_old_and_candidate(self):
        self.target.write_bytes(self.old.replace(b'"format_version": 1', b'"format_version": 9'))
        result = self.call(expected_sha256=hashlib.sha256(self.target.read_bytes()).hexdigest())
        self.assertEqual(result['status'], 'unknown', result)
        self.target.write_bytes(self.old)
        self.assertEqual(self.call(candidate_bytes=self.new.replace(b'"format_version": 1', b'"format_version": 9'))['status'], 'pending')

    def test_receipt_without_entry_rejected(self):
        receipt = dict(delta_id='task/content-creator/run/1', sha256='1' * 64,
                       result_revision=1, entry_ids=['missing'])
        result = self.call(candidate_bytes=encoded(1, receipts=[receipt]))
        self.assertEqual(result['status'], 'pending', result)
        self.assertFalse((self.root / (self.rel + '.pending')).exists())

    def test_stale_target_rechecked_after_staging(self):
        real_fsync = os.fsync
        changed = encoded(2)
        calls = []
        def mutate_once(fd):
            calls.append(fd)
            if len(calls) == 1:
                self.target.write_bytes(changed)
            return real_fsync(fd)
        with patch.object(saver.os, 'fsync', side_effect=mutate_once):
            result = self.call()
        self.assertEqual(result['status'], 'pending', result)
        self.assertEqual(self.target.read_bytes(), changed)
        self.assertEqual((self.root / (self.rel + '.pending')).read_bytes(), self.new)

    def test_pre_replace_failure_keeps_original_and_pending(self):
        with patch.object(saver.os, 'replace', side_effect=OSError(5, 'injected failure')):
            result = self.call()
        self.assertEqual(result['status'], 'pending', result)
        self.assertEqual(self.target.read_bytes(), self.old)
        self.assertEqual((self.root / (self.rel + '.pending')).read_bytes(), self.new)

    def test_real_record_roundtrip_and_archive(self):
        source = BASE.parent.parent / 'input/project/.codex/agent-memory/content-creator.md'
        original = source.read_bytes()
        record = saver._parse(original)
        record['history'].append(dict(revision=record['revision'], entries=record['entries'], receipts=record['receipts']))
        record['revision'] += 1
        candidate = ('# 完整记忆\n\n```json\n' + json.dumps(record, ensure_ascii=False, indent=2) + '\n```\n').encode()
        self.target.write_bytes(original)
        result = self.call(candidate_bytes=candidate, expected_sha256=hashlib.sha256(original).hexdigest())
        self.assertEqual(result['status'], 'saved', result)
        rel = 'archive/content-creator.md'
        result = self.call(target_rel=rel, candidate_bytes=candidate, expected_sha256=None,
                           allowed_paths={rel, rel + '.pending'})
        self.assertEqual(result['status'], 'saved', result)
        self.assertEqual((self.root / rel).read_bytes(), candidate)
        self.assertEqual(source.read_bytes(), original)

    def test_file_sync_failure_keeps_original(self):
        with patch.object(saver.os, 'fsync', side_effect=OSError(5, 'injected file sync failure')):
            result = self.call()
        self.assertEqual(result['status'], 'pending', result)
        self.assertEqual(result['sync']['file'], 'failed', result)
        self.assertEqual(self.target.read_bytes(), self.old)
        self.assertEqual((self.root / (self.rel + '.pending')).read_bytes(), self.new)

    def test_interruption_before_replace_keeps_complete_original(self):
        with patch.object(saver.os, 'replace', side_effect=KeyboardInterrupt):
            with self.assertRaises(KeyboardInterrupt):
                self.call()
        self.assertEqual(self.target.read_bytes(), self.old)
        self.assertEqual((self.root / (self.rel + '.pending')).read_bytes(), self.new)

    def test_unverifiable_target_after_replace_is_unknown(self):
        real_replace = os.replace
        def replace_then_damage(src, dst):
            real_replace(src, dst)
            self.target.write_bytes(b'broken after commit')
        with patch.object(saver.os, 'replace', side_effect=replace_then_damage):
            result = self.call()
        self.assertEqual(result['status'], 'unknown', result)
        self.assertIsNone(result['committed'])
        self.assertEqual(self.target.read_bytes(), b'broken after commit')

    def test_directory_sync_failure_rechecks_committed_target(self):
        real_fsync = os.fsync
        calls = []
        def fail_directory(fd):
            calls.append(fd)
            if len(calls) > 1:
                raise OSError(5, 'injected directory sync failure')
            return real_fsync(fd)
        with patch.object(saver.os, 'fsync', side_effect=fail_directory):
            result = self.call()
        self.assertEqual(result['status'], 'saved', result)
        self.assertEqual(result['sync']['directory'], 'failed', result)
        self.assertEqual(self.target.read_bytes(), self.new)

    def test_exception_after_committing_replace_does_not_report_unsaved(self):
        real_replace = os.replace
        def commit_then_raise(src, dst):
            real_replace(src, dst)
            raise OSError(5, 'injected lost completion')
        with patch.object(saver.os, 'replace', side_effect=commit_then_raise):
            result = self.call()
        self.assertEqual(result['status'], 'saved', result)
        self.assertEqual(self.target.read_bytes(), self.new)

if __name__ == '__main__':
    unittest.main(verbosity=2)
